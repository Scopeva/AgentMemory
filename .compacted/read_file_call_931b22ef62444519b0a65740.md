    51	    project_id: Optional[str] = None,

    57	    from agent_memory.core.projects import identity
    58	
    59	    if project_id:
    60	        result = await identity.resolve_project_by_id(db, project_id)
    61	        if result.project is None:
    62	            raise ProjectUnresolvedError(result.reason or "Project not found")
    63	        return {
    64	            "project": result.project,
    65	            "workspace": result.workspace,
    66	            "repository": result.repository,
    67	            "status": result.status,
    68	            "reason": result.reason,
    69	        }
    70	
    71	    if workspace_path:
    72	        result = await identity.resolve_project_by_workspace(db, workspace_path)
    73	        if result.status == "unresolved":
    74	            raise ProjectUnresolvedError(result.reason or "Workspace not registered")
    75	        # Resolve repository and project from workspace
    76	        project = None
    77	        repository = None
    78	        if result.workspace:
    79	            # Look up repository by workspace.repository_id
    80	            async with db.connection() as conn:
    81	                cursor = await conn.execute(
    82	                    "SELECT repository_id, project_id FROM workspaces WHERE workspace_id = ?",
    83	                    (result.workspace.workspace_id,),
    84	                )
    85	                row = await cursor.fetchone()
    86	                if row:
    87	                    repository = await repo.projects.get_repository_by_fingerprint(
    88	                        db, row["project_id"], ""
    89	                    ) if False else None
    90	                    # Simpler: fetch project via workspace -> repository join
    91	                    cursor2 = await conn.execute(
    92	                        "SELECT p.project_id, p.name, p.status "
    93	                        "FROM projects p "
    94	                        "JOIN repositories r ON r.project_id = p.project_id "
    95	                        "JOIN workspaces w ON w.repository_id = r.repository_id "
    96	                        "WHERE w.workspace_id = ?",
    97	                        (result.workspace.workspace_id,),
    98	                    )
    99	                    prow = await cursor2.fetchone()
   100	                    if prow:
   101	                        project = Project(
   102	                            project_id=prow["project_id"],
   103	                            name=prow["name"],
   104	                            status=prow["status"],
   105	                        )

   107	            "project": project,
   108	            "workspace": result.workspace,
   109	            "repository": repository,
   110	            "status": "resolved" if project else "partial",
