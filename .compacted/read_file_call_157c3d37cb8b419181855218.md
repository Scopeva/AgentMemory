    27	async def find_superseded(
    28	    db: Database,
    29	    project_id: str,
    30	    content_threshold: float = 0.85,
    31	    limit: int = 50,
    32	) -> list[dict]:
    33	    """Find accepted knowledge items that may have been superseded.
    34	
    35	    Heuristic: two accepted items with the same (project, type) where the
    36	    newer one shares >= content_threshold similarity with the older one.
    37	    """
    38	    items = await repo.knowledge.list_knowledge_by_project(
    39	        db, project_id, status=KnowledgeStatus.ACCEPTED, limit=limit * 4
    40	    )
    41	    # Group by type
    42	    by_type: dict[str, list[Knowledge]] = {}
    43	    for k in items:
    44	        by_type.setdefault(k.type, []).append(k)
    45	
    46	    results: list[dict] = []
    47	    for ktype, group in by_type.items():
    48	        group.sort(key=lambda k: k.updated_at)
    49	        for i in range(len(group) - 1):
    50	            older, newer = group[i], group[i + 1]
    51	            sim = _similarity(older.content, newer.content)
    52	            if sim >= content_threshold:
    53	                results.append({
    54	                    "superseded_memory_id": older.memory_id,
    55	                    "superseded_title": older.title,
    56	                    "supersedes_memory_id": newer.memory_id,
    57	                    "supersedes_title": newer.title,
    58	                    "type": ktype,
    59	                    "similarity": round(sim, 3),
    60	                })
