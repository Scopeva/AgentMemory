    12	def _utcnow() -> str:
    13	    return datetime.now(timezone.utc).replace(tzinfo=None).isoformat() + "Z"
    14	
    15	
    16	async def register_project(db: Database, project: Project) -> Project:
    17	    now = _utcnow()
    18	    async with db.connection() as conn:
    19	        await conn.execute(
    20	            "INSERT INTO projects (project_id, name, status, created_at, updated_at) "
    21	            "VALUES (?, ?, ?, ?, ?)",
    22	            (project.project_id, project.name, project.status.value, now, now),
    23	        )
    24	        await conn.commit()
    25	    return project
    26	
    27	
    28	async def get_project(db: Database, project_id: str) -> Optional[Project]:
    29	    async with db.connection() as conn:
    30	        cursor = await conn.execute(
    31	            "SELECT project_id, name, status, created_at, updated_at "
    32	            "FROM projects WHERE project_id = ?",
    33	            (project_id,),
    34	        )
    35	        row = await cursor.fetchone()
    36	        if not row:
    37	            return None
    38	        return Project(
    39	            project_id=row["project_id"],
    40	            name=row["name"],
    41	            status=row["status"],
    42	            created_at=row["created_at"],
    43	            updated_at=row["updated_at"],
    44	        )
