   267	async def submit_memory_candidate(
   268	    db: Database,
   269	    project_id: str,
   270	    idempotency_key: str,
   271	    memories: list[dict],
   272	    run_id: Optional[str] = None,
   273	) -> dict:
   274	    """Submit memory candidates. Idempotent per key; one candidate per memory."""
   275	    inserted: list[str] = []
   276	    duplicates = 0
   277	    for memory in memories:
   278	        new = await repo.knowledge.insert_memory_candidate(
   279	            db, project_id, run_id, idempotency_key=f"{idempotency_key}:{memory.get('title','')}",
   280	            memory_json=memory,
   281	        )
   282	        if new:
   283	            inserted.append(str(memory.get("memory_id", "")))
   284	        else:
   285	            duplicates += 1
   286	    return {
   287	        "candidate_ids": inserted,
   288	        "validation_status": ["ok" for _ in inserted] + ["duplicate" for _ in range(duplicates)],
   289	        "duplicate_count": duplicates,
   290	    }
